/** The pieces, in a host thread — no port, no banner. */
export function createModule(options = {}) {
  const dir = path.resolve(options.dir ?? './services');
  const catalog = createServiceCatalog({ dir });
  return { catalog, registry: createServiceMcpRegistry(catalog), dir };
}

/** This module's own little door on a port (port 0 = OS-assigned). */
export async function startModule(options = {}) {
  const dir = path.resolve(options.dir ?? './services');
  const catalog = createServiceCatalog({ dir });
  const api = await startApiServer({
    catalog,
    port: options.port ?? 8128,
    host: options.host ?? '127.0.0.1',
    token: options.token ?? null,
    publicDir: options.publicDir ?? path.join(path.dirname(fileURLToPath(import.meta.url)), 'public'),
  });
  return {
    catalog,
    registry: createServiceMcpRegistry(catalog),
    api,
    url: api.url,
    port: api.port,
    dir,
    stop: () => api.stop(),
  };
}

// Drives the REAL thing — a real service file, a real port — and sets process.exitCode
// (never process.exit). Same spirit as every module's runTests, at the entry level.

async function selfCheck() {
  const fs = await import('node:fs');
  const os = await import('node:os');
  let failures = 0;
  const check = (label, cond) => {
    console.log(`${cond ? 'pass' : 'FAIL'} ${label}`);
    if (!cond) failures++;
  };

  const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'svc-entry-'));
  fs.writeFileSync(
    path.join(dir, 'demo.js'),
    'module.exports = { name: "demo", advert: { what: "a demo", karma: 1 }, provider: { name: "demo app" }, modes: [{ name: "cpu", install: { steps: ["do it"] } }], pages: { ad: "<h1>the demo page</h1>", inputs: "<h1>in</h1>" } };'
  );

  const pieces = createModule({ dir });
  check('createModule: the catalog lists the file', pieces.catalog.list().length === 1 && pieces.catalog.get('demo') !== null);
  check('createModule: the registry advertises six tools', pieces.registry.list().length === 6);

  const running = await startModule({ dir, port: 0 });
  const health = await (await fetch(running.url + '/health')).json();
  check('startModule: health answers', health.ok === true && health.services === 1);
  const listing = await (await fetch(running.url + '/api/services')).json();
  check('startModule: the listing answers', Array.isArray(listing.services) && listing.services[0].name === 'demo');
  const ad = await (await fetch(running.url + '/api/services/demo/ad')).text();
  check('startModule: the ad page answers', ad.includes('the demo page'));
  const inPage = await (await fetch(running.url + '/api/services/demo/inputs')).text();
  check('startModule: the inputs page answers', inPage.includes('<h1>in</h1>'));
  const init = await (
    await fetch(running.url + '/mcp', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ jsonrpc: '2.0', id: 1, method: 'initialize', params: {} }),
    })
  ).json();
  check('startModule: the agent door answers', init.result?.serverInfo?.name === 'services');
  await running.stop();
  let closed = false;
  try {
    await fetch(running.url + '/health');
  } catch {
    closed = true;
  }
  check('stop closes the door', closed);

  console.log(failures === 0 ? '[Services_module] self-check GREEN' : `[Services_module] ${failures} FAILED`);
  process.exitCode = failures === 0 ? 0 : 1;
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  void selfCheck();
}

 *              catalog carries the refusal, nothing crashes.
 * @intent Operator ruling (2026-09-23): "`.js` is best I think too" — one .js file per
 *         service; and the tie (2026-09-23): "a shared service file must not execute on
 *         receipt — load it as DATA." Heavy items travel by LINK, never inside the file;
 *         small images inline as base64; the pages ride as strings (write page scripts
 *         with string concatenation — a literal `${` is refused, and nested backticks end
 *         the literal). Honest limits: the token gate refuses the obvious, the sandbox is
 *         the real door — even a file that slipped past every gate would be running in an
 *         empty 2-second room with no doors, and its result must still be pure data.
 */
import { type ServiceDefinition, type ValidationReport } from './serviceSchema.js';
export interface ServiceFileResult {
    /** The file that was read. */
    file: string;
    /** The loaded definition (present only when everything passed). */
    definition: ServiceDefinition | null;
    /** The schema report (errors/warnings) — present even when refused early. */
    report: ValidationReport;
    /** The refusal reason, when the file never became data. */
    error: string | null;
}
/** Tokens that are code, not data — refused anywhere in the text. */
export declare const CODE_TOKENS: string[];
/** Walk the loaded value: functions, symbols, bigints and cycles are refused with a path. */
export declare function findRefusedValue(value: unknown, pathStr?: string, seen?: Set<unknown>, depth?: number): string | null;
/** The whole rung-ladder over TEXT (no fs) — the testable heart. */
export declare function loadServiceText(text: string): {
    value: unknown;
    error: string | null;
};
/** Read one file from disk and carry it through every rung. */
export declare function loadServiceFile(filePath: string): ServiceFileResult;
export declare function runTests(): Promise<void>;

/**
 * @module serviceSchema
 * @description THE SERVICE OBJECT — the shape of one service definition (the elaborate
 *              object as the talk settled it): advert (what · expected time · karma weight)
 *              · provider (the app that runs it, with install/update/launch/probe) ·
 *              variables (init · input · files — a file slot plugs a file OR a service that
 *              produces it) · modes (one service, several fits — each with its own models,
 *              needs, karma weight and INSTALL, where the version dance lives: pinned
 *              python·cuda·torch combos and per-sm_XX build notes for older cards) · needs
 *              (what the Resource Manager books) · output · children (declared, wired by
 *              their outputs) · update (a declared command) · pages (the TWO the service
 *              PRODUCES: ad — the sell sheet; inputs — the way in, which gathers the
 *              required inputs and carries progress during the run) · progress (the
 *              declared how-to — the INSTANCE tracks and shows it) · gui (the service's own
 *              gui, when it has one — a button follows it). v0.2: the core is checked hard,
 *              everything else passes through untouched — nothing frozen; the object stays
 *              liquid.
 * @intent "Services is just an elaborate object" (operator). Services declares; nothing here
 *         runs. A service file is DATA — validated, never executed.
 */
export interface ServiceVariable {
    name: string;
    label?: string;
    type?: 'text' | 'textarea' | 'number' | 'select' | 'boolean' | 'range';
    default?: unknown;
    required?: boolean;
    min?: number;
    max?: number;
    step?: number;
    options?: string[];
    help?: string;
    full?: boolean;
    muted?: boolean;
    /** 'output' marks the output path among the inputs — give it a default; the instance keeps filenames fresh (never the same name twice when the user is lazy). */
    role?: string;
    [extra: string]: unknown;
}
/** A required FILE — per request (a run's input file), or per install (a node's file). */
export interface ServiceFileSlot {
    name: string;
    path?: string;
    kind?: string;
    note?: string;
    /** The slot may plug a FILE, or a SERVICE that produces it. */
    service?: string;
    [extra: string]: unknown;
}
/** A LINK — everything heavy travels by link, never inside the file. */
export interface ServiceLink {
    path?: string;
    url?: string;
    sha256?: string;
    source?: {
        magnet?: string;
        peer?: string;
        [extra: string]: unknown;
    };
    note?: string;
    [extra: string]: unknown;
}
/** THE INSTALL FACE — where the version dance lives (older GPU cards are first-class). */
export interface ServiceInstall {
    /** Pinned toolchain combos that still exist and fit: { python: '3.11', cuda: '11.8', torch: '2.5.1+cu118' }. */
    pins?: Record<string, string>;
    /** The card's compute, when a build is per-architecture: e.g. '52' → an sm_52 build. */
    sm?: string;
    arch?: string;
    /** How it is installed, in words/steps the installer can follow. */
    steps?: string[];
    /** Where the payload comes from when the network cannot serve it (paths, urls). */
    fallback?: string[];
    note?: string;
    [extra: string]: unknown;
}
/** THE TWO PAGES a service produces — the sell sheet and the way in. */
export interface ServicePages {
    /** The AD page — the sell sheet (the listing): what it does in the maker's own voice · images (small ones inline as base64) · whatever sells it. */
    ad: string;
    /** The INPUTS page — the way in: gathers the required inputs before the run; during the run the progress rides on it. Intentionally vague and custom — made with the user. */
    inputs: string;
}
/** HOW PROGRESS IS SEEN — declared here; the INSTANCE (Basic_App) tracks and shows it. */
export interface ServiceProgress {
    /** In words: how can anyone tell how far along a run is (tokens stream · "step N of M" lines · silent until the end). */
    how: string;
    /** Where it reads from when known: stdout · file · http · none. */
    source?: string;
    /** A pattern or key to look for, when there is one. */
    pattern?: string;
    note?: string;
    [extra: string]: unknown;
}
/** THE SERVICE'S OWN GUI, when it has one — Basic_App shows a button that follows it. */
export interface ServiceGui {
    /** How to reach it: url (a web face it serves) · exe (an app that opens) · command · other. */
    kind?: string;
    /** The url, path or command. */
    value?: string;
    note?: string;
    [extra: string]: unknown;
}
/** ONE MODE — the fit between the service and a machine: a smaller mode is how a service reaches smaller nodes. */
export interface ServiceMode {
    name: string;
    /** What this mode fits: { vramGbMin: '6', sm: '<=52', note: '…' }. */
    fits?: Record<string, string>;
    models?: ServiceLink[];
    needs?: ServiceNeeds;
    /** Karma weight for this mode — the service's own number; the ledger lives elsewhere. */
    karma?: number;
    install?: ServiceInstall;
    [extra: string]: unknown;
}
/** What the Resource Manager will book for a run. */
export interface ServiceNeeds {
    gpu?: boolean;
    vramGb?: number;
    ramMb?: number;
    cores?: number;
    diskGb?: number;
    services?: string[];
    [extra: string]: unknown;
}
export interface ServiceDefinition {
    /** Unique in the catalog (the folder IS the catalog). */
    name: string;
    version?: string;
    /** what it does · expected time (HISTORY: average or last runs — unknown is unknown) · karma weight. */
    advert: {
        what: string;
        time?: string;
        karma?: number;
        note?: string;
        [extra: string]: unknown;
    };
    /** THE APP THAT RUNS IT — the provider lives INSIDE the service definition. */
    provider: {
        name: string;
        repo?: string;
        launch?: string;
        probe?: string;
        update?: string;
        note?: string;
        [extra: string]: unknown;
    };
    /** init = install-time facts · input = per-request values · files = per-request required files. */
    variables?: {
        init?: ServiceVariable[];
        input?: ServiceVariable[];
        files?: ServiceFileSlot[];
        [extra: string]: unknown;
    };
    modes?: ServiceMode[];
    needs?: ServiceNeeds;
    output?: {
        kind?: string;
        location?: string;
        description?: string;
        [extra: string]: unknown;
    };
    /** Child services, declared by name — their outputs wire into this service's file slots. */
    children?: string[];
    /** Default install when a mode carries none. */
    install?: ServiceInstall;
    /** The declared update: HOW (a command); WHEN is future wisdom. */
    update?: {
        command?: string;
        note?: string;
        [extra: string]: unknown;
    };
    /** THE TWO PAGES a service produces: pages.ad (the sell sheet) · pages.inputs (the way in — gathers the required inputs; carries progress during the run). Both are the maker's, carried as strings. */
    pages: ServicePages;
    /** HOW PROGRESS IS SEEN for a run — declared here; the INSTANCE (Basic_App) tracks and shows it. Absent = unknown (the instance falls back honestly). */
    progress?: ServiceProgress;
    /** THE SERVICE'S OWN GUI, when it has one — a button follows it. Absent = no gui of its own. */
    gui?: ServiceGui;
    [extra: string]: unknown;
}
export interface ValidationIssue {
    level: 'error' | 'warning';
    field: string;
    message: string;
}
export interface ValidationReport {
    ok: boolean;
    errors: ValidationIssue[];
    warnings: ValidationIssue[];
}
/** The SHAPE that a listing carries — contents only ever answer a get. */
export interface ServiceSummary {
    name: string;
    what: string;
    provider: string;
    modes: string[];
    karma: number | null;
    output: string | null;
    children: number;
    ok: boolean;
    file: string;
}
/** Check one service definition. Errors refuse it; warnings ride along. */
export declare function validateService(value: unknown): ValidationReport;
/** The listing shape of one definition — names and counts only, never the page. */
export declare function summarizeService(def: ServiceDefinition, file: string, ok: boolean): ServiceSummary;
