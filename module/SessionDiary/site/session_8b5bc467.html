"""Print the title bars of all visible top-level windows on Windows.

Usage: python list_windows.py

Uses only the standard library (ctypes) - no installs needed.
"""
import ctypes
from ctypes import wintypes

user32 = ctypes.windll.user32

def get_window_titles():
"""Enumerate all visible top-level windows and return their titles."""
titles = []

def callback(hwnd, _):
# Only count windows that are actually visible
if user32.IsWindowVisible(hwnd):
length = user32.GetWindowTextLengthW(hwnd)
if length > 0:
buf = ctypes.create_unicode_buffer(length + 1)
user32.GetWindowTextW(hwnd, buf, length + 1)
title = buf.value.strip()
if title:
titles.append(title)
return True # keep enumerating

WNDENUMPROC = ctypes.WINFUNCTYPE(wintypes.BOOL, wintypes.HWND, wintypes.LPARAM)
user32.EnumWindows(WNDENUMPROC(callback), 0)
return titles

def main():
titles = get_window_titles()
for t in titles:
print(t)
print(f"\n({len(titles)} window{'s' if len(titles) != 1 else ''} found)")

if __name__ == "__main__":
main()

**Implementation notes:**
- **Focus** uses `ShowWindow(SW_RESTORE)` + `SetForegroundWindow`, with an `AttachThreadInput` fallback to get past Windows' foreground-lock rule if the direct call is refused.
- **Keyboard** is pure `ctypes` `SendInput` (proper `INPUT`/`KEYBDINPUT` struct definitions, explicit `argtypes` so 64-bit handles don't truncate) — no pyautogui/dependencies.
- **Clipboard** is pure ctypes: `OpenClipboard` → `EmptyClipboard` → `GlobalAlloc`/`Lock`/`Unlock` with null-terminated UTF-16 → `SetClipboardData(CF_UNICODETEXT)` → `CloseClipboard` in a `finally`. Non-ASCII text works.
- Settle delays: 500 ms after focus (as specified), 300 ms after Ctrl+N (new window to come up), 300 ms after paste (text to land) before Enter — all constants at the top of the file if you want to tune them.

One heads-up: the title match is case-insensitive on "hermes" and picks the *first* matching visible window — if you ever have two Hermes windows open, the first one found gets the keystrokes.
