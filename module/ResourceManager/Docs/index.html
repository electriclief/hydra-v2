Purpose

Resource Manager — the register. Reads this node's resources (cpu cores · gpus with vram truth · ram · disks · Hydra shared folders · observed gpu-side processes) and keeps BOOKINGS: what a run holds (amounts for cores/ram/disk, a gpu whole+pinned, paths by subtree). No internal logic — poll is a pure read, book the one write, pop/clear the exits; the only rule is it won't allow double booking (1 app per gpu). Not enforcement — voluntary all the way.

/** The pieces, in a host thread — no port, no banner. */
export function createModule(options = {}) {
  const dataDir = path.resolve(options.dataDir ?? './data');
  const sharesFile = options.sharesFile ?? path.join(dataDir, 'shares.json');
  const collect = { shares: loadShareList(sharesFile) };
  const reads = options.reads ?? (() => collectReads(collect));
  const store = createBookingStore({ dataDir, reads });
  return { store, reads, registry: createRmMcpRegistry(store, reads), dataDir, sharesFile };
}

/** This module's own little door on a port (port 0 = OS-assigned). */
export async function startModule(options = {}) {
  const dataDir = path.resolve(options.dataDir ?? './data');
  const sharesFile = options.sharesFile ?? path.join(dataDir, 'shares.json');
  const collect = { shares: loadShareList(sharesFile) };
  const reads = options.reads ?? (() => collectReads(collect));
  const refreshReads = options.reads ?? (() => collectReads({ ...collect, force: true }));
  const store = createBookingStore({ dataDir, reads });
  const api = await startApiServer({
    store,
    reads,
    refreshReads,
    port: options.port ?? 8129,
    host: options.host ?? '127.0.0.1',
    token: options.token ?? null,
    publicDir: options.publicDir ?? path.join(path.dirname(fileURLToPath(import.meta.url)), 'public'),
  });
  return {
    store,
    reads,
    registry: createRmMcpRegistry(store, reads),
    api,
    url: api.url,
    port: api.port,
    dataDir,
    stop: () => api.stop(),
  };
}

// Drives the REAL thing — a real store, a real port — and sets process.exitCode
// (never process.exit). Same spirit as every module's runTests, at the entry level.

async function selfCheck() {
  const fs = await import('node:fs');
  const os = await import('node:os');
  let failures = 0;
  const check = (label, cond) => {
    console.log(`${cond ? 'pass' : 'FAIL'} ${label}`);
    if (!cond) failures++;
  };

  const fakeReads = {
    collectedAt: new Date().toISOString(),
    node: 'entry-test',
    platform: 'win32',
    cpu: { model: 'fake', cores: 8, loadPct: 1 },
    ram: { totalMb: 16000, freeMb: 9000 },
    gpus: [{ name: 'RX 6800', vramMb: 16384, vramSource: 'registry', memUsedMb: null }],
    gpusObserved: [],
    disks: [{ device: 'C:', freeMb: 100000, totalMb: 500000 }],
    shares: [],
    notes: [],
  };
  const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'rm-entry-'));

  const pieces = createModule({ dataDir: dir, reads: () => fakeReads });
  check('createModule: the registry advertises five tools', pieces.registry.list().length === 5);
  const booked = await pieces.store.book({ service: 'demo', gpu: 'RX 6800', cores: 2 });
  check('createModule: the register books for real', booked.ok === true && booked.booking.id.startsWith('bk-'));
  const refused = await pieces.store.book({ service: 'other', gpu: 'RX 6800' });
  check('createModule: double booking is refused with the holder', refused.ok === false && refused.error.blockers[0].heldBy?.service === 'demo');

  const running = await startModule({ dataDir: dir, reads: () => fakeReads, port: 0 });
  const health = await (await fetch(running.url + '/health')).json();
  check('startModule: health answers', health.ok === true && health.bookings === 1);
  const poll = await (await fetch(running.url + '/api/resources')).json();
  check('startModule: the poll answers', poll.ok === true && poll.resources.cpu.cores === 8 && poll.bookings.length === 1);
  const bookViaHttp = await (
    await fetch(running.url + '/api/book', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ service: 'via-http', cores: 1 }) })
  ).json();
  check('startModule: book answers over HTTP', bookViaHttp.ok === true && bookViaHttp.booking.service === 'via-http');
  const init = await (
    await fetch(running.url + '/mcp', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ jsonrpc: '2.0', id: 1, method: 'initialize', params: {} }),
    })
  ).json();
  check('startModule: the agent door answers', init.result?.serverInfo?.name === 'resource-manager');
  await running.stop();
  let closed = false;
  try {
    await fetch(running.url + '/health');
  } catch {
    closed = true;
  }
  check('stop closes the door', closed);

  console.log(failures === 0 ? '[ResourceManager_module] self-check GREEN' : `[ResourceManager_module] ${failures} FAILED`);
  process.exitCode = failures === 0 ? 0 : 1;
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  void selfCheck();
}

 * @description THE RESOURCE MODEL — every shape this register speaks, in one place:
 *              the READS (cpu · ram · gpus + observed processes · disks · Hydra shared

 *         two exits; the only rule is it won't allow double booking (1 app per gpu).

 *         don't step on each other. Nothing here is enforcement — voluntary all the way.

/** One gpu, as read on this node. vram truth: nvidia-smi > driver registry (the
 *  AdapterRAM UInt32 caps at 4095 MB) > AdapterRAM > unknown. */
export interface GpuRead {
    name: string;

    vramSource: 'nvidia-smi' | 'registry' | 'adapterram' | 'unknown';
    memUsedMb: number | null;
}
/** A process this node can SEE on the gpu side that nobody booked — information only,
 *  never a rule (the V1 lesson: llama started outside the runner still counts). */
export interface GpuObserved {

    pid: number;
    reason: string;
}
export interface DiskRead {
    device: string;
    freeMb: number;
    totalMb: number;
}
/** A Hydra shared folder on this node (space-gated: it exists when the disk allows).
 *  The register READS these — it does not create or manage them. */
export interface ShareRead {
    label: string | null;
    path: string;
    exists: boolean;
    freeMb: number | null;
    totalMb: number | null;
}
/** THE POLL — a pure read. Asking holds nothing and remembers nothing. */
export interface ResourceReads {
    collectedAt: string;
    node: string;
    platform: string;
    cpu: {
        model: string;
        cores: number;
        loadPct: number | null;
    };
    ram: {
        totalMb: number;
        freeMb: number;
    };
    gpus: GpuRead[];
    gpusObserved: GpuObserved[];
    disks: DiskRead[];
    shares: ShareRead[];
    notes: string[];
}
/** A book request — the CALLER chose these resources; the register only checks them. */
export interface BookRequest {
    /** The service this run is (the booking's identity — the scheduler points at it). */
    service: string;
    mode?: string | null;
    /** A gpu by name, exactly as the poll shows it. Whole and pinned — 1 app per gpu. */
    gpu?: string | null;
    cores?: number | null;
    ramMb?: number | null;
    diskMb?: number | null;
    /** Folders this run works in. A booked folder covers its subtree. */
    paths?: string[];
    /** The scheduler's own expectation (recorded as info; null = unknown). */
    expected?: string | null;
    note?: string | null;
    /** Optional run id, when the caller carries one. */
    run?: string | null;
}
export interface Booking {
    id: string;
    service: string;
    mode: string | null;
    gpu: string | null;
    cores: number | null;
    ramMb: number | null;
    diskMb: number | null;
    paths: string[];
    expected: string | null;
    note: string | null;
    run: string | null;
    bookedAt: string;

export interface EndedBooking extends Booking {

export interface BookingBlocker {
    kind: 'gpu' | 'cores' | 'ram' | 'disk' | 'path';

        id: string;
        service: string;
        since: string;
    } | null;
}
export interface BookRefusal {
    ok: false;
    error: {
        code: 'double_booked' | 'missing_resource' | 'invalid_book' | 'no_such_booking';
        message: string;
        blockers: BookingBlocker[];

}
export interface BookSuccess {
    ok: true;
    booking: Booking;
}
